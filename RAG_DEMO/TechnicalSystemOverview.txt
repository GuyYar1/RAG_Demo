tokenizer = AutoTokenizer.from_pretrained("BioMistral-7B")  # Replace with your model's tokenizer

def chunk_text(text, chunk_size_tokens=512, overlap_tokens=50):
    # Tokenize the text
    tokens = tokenizer.encode(text)  # Converts the text into tokens
    
    chunks = []
    start = 0
    while start < len(tokens):
        # End token index for the chunk
        end = min(start + chunk_size_tokens, len(tokens))
        
        # Get the token chunk
        chunk_tokens = tokens[start:end]
        
        # Convert token chunk back to text
        chunk_text = tokenizer.decode(chunk_tokens, skip_special_tokens=True)
        
        chunks.append(chunk_text)
        
        # Move the start index, applying overlap
        start = end - overlap_tokens  # Overlap ensures continuity
    
    return chunks
